cmake_minimum_required(VERSION 3.30)
project(small_llt LANGUAGES CXX)
set(scan_controls c_scan2 c_scan4 c_flat_scan2 c_flat_scan4 c_toltile c_flat_toltile c_combined_scan4_tol)
set(EIGEN_SOURCE_DIR "" CACHE PATH "Pinned Eigen")
foreach(target before_full after_full)
  add_executable(${target} full.cpp)
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(target STREQUAL "before_full")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/variants/baseline")
  else()
    target_include_directories(${target} PRIVATE ../../../include)
  endif()
  target_include_directories(${target} PRIVATE ../../../tests "${EIGEN_SOURCE_DIR}")
  target_compile_definitions(${target} PRIVATE EIGEN_DONT_PARALLELIZE=1)
  target_compile_options(${target} PRIVATE /O2 /fp:precise)
endforeach()
foreach(variant baseline fused inline_input fused_inline panel16 fused_panel16 all16 dot fused_dot width4 width16 leftlooking left_inline left_reciprocal left_vector left_unroll left_pair left_nocopy left_force left_small left_sym left_backward left_copy left_restore left_combined c_current c_sum c_six c_sum_six c_diagsum c_diagpair c_diagcopy c_sum_diagcopy c_flat c_no_finite c_no_symmetry c_no_validation c_symtile c_flat_symtile ${scan_controls})
  add_executable(${variant} ../assembly-gap/phases.cpp)
  target_compile_features(${variant} PRIVATE cxx_std_20)
  target_include_directories(${variant} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/variants/${variant}" ../../../tests "${EIGEN_SOURCE_DIR}")
  target_compile_definitions(${variant} PRIVATE EIGEN_DONT_PARALLELIZE=1)
  target_compile_options(${variant} PRIVATE /O2 /fp:precise /FAs "/Fa${CMAKE_CURRENT_BINARY_DIR}/${variant}.asm")
endforeach()
add_executable(before_numerical ../../../tests/numerical_tests.cpp)
target_compile_features(before_numerical PRIVATE cxx_std_20)
target_include_directories(before_numerical PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/variants/baseline" "${EIGEN_SOURCE_DIR}")
target_compile_options(before_numerical PRIVATE /O2 /fp:precise)
foreach(target actual actual_solver before_solver)
  if(target STREQUAL "actual")
    add_executable(${target} ../assembly-gap/phases.cpp)
  else()
    add_executable(${target} ../solver-locality/probe.cpp)
  endif()
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(target STREQUAL "before_solver")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/variants/baseline")
  else()
    target_include_directories(${target} PRIVATE ../../../include)
  endif()
  target_include_directories(${target} PRIVATE ../../../tests "${EIGEN_SOURCE_DIR}")
  target_compile_definitions(${target} PRIVATE EIGEN_DONT_PARALLELIZE=1)
  target_compile_options(${target} PRIVATE /O2 /fp:precise /FAs "/Fa${CMAKE_CURRENT_BINARY_DIR}/${target}.asm")
endforeach()
