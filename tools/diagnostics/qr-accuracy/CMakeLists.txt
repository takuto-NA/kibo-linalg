cmake_minimum_required(VERSION 3.30)
project(qr_accuracy LANGUAGES CXX)
set(EIGEN_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../.cache/eigen" CACHE PATH "Pinned Eigen headers")
option(KIBO_DIAGNOSTIC_SCALAR "Disable explicit and automatic vectorization in both backends" OFF)
add_executable(audit audit.cpp)
if(MSVC)
  add_executable(cost cost.cpp)
  add_executable(public-cost public-cost.cpp)
  set(targets audit cost public-cost)
else()
  set(targets audit)
endif()
foreach(target IN LISTS targets)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE ../../../include ../../../tests "${EIGEN_SOURCE_DIR}")
  target_compile_definitions(${target} PRIVATE EIGEN_DONT_PARALLELIZE=1)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /fp:precise)
  else()
    target_compile_options(${target} PRIVATE -O2 -ffp-contract=off)
  endif()
  if(KIBO_DIAGNOSTIC_SCALAR)
    if(MSVC)
      message(FATAL_ERROR "Pinned MSVC ignores /Qvec-; use GCC or Clang for scalar comparison")
    endif()
    target_compile_definitions(${target} PRIVATE KIBO_DISABLE_SIMD=1 EIGEN_DONT_VECTORIZE=1)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
      target_compile_options(${target} PRIVATE -fno-vectorize -fno-slp-vectorize)
    else()
      target_compile_options(${target} PRIVATE -fno-tree-vectorize -fno-tree-slp-vectorize)
    endif()
  endif()
endforeach()
