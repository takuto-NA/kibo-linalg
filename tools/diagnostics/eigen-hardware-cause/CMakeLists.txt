cmake_minimum_required(VERSION 3.30)
project(eigen_hardware_cause LANGUAGES CXX)
set(EIGEN_SOURCE_DIR "" CACHE PATH "Verified pinned Eigen archive headers")
function(probe name owner source)
  add_executable(${name} ${source})
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_include_directories(${name} PRIVATE ../eigen-root-cause/${owner}/include ../eigen-gap/include ../eigen-gap/tests "${EIGEN_SOURCE_DIR}" ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(${name} PRIVATE EIGEN_DONT_PARALLELIZE=1 ${ARGN})
  target_compile_options(${name} PRIVATE /O2 /fp:precise)
endfunction()
probe(stride llt stride_probe.cpp)
probe(read_stride llt read_stride_probe.cpp)
probe(projection qr projection_probe.cpp)
probe(llt_guarded_column llt llt_validation_probe.cpp KIBO_DIAG_LLT_COLUMN_WIDTH=8)
probe(llt_no_symmetry llt llt_validation_probe.cpp KIBO_DIAG_LLT_COLUMN_WIDTH=8 CAUSE_NO_SYMMETRY=1)
foreach(mode cpp sse)
  add_library(asm_${mode} OBJECT assembly_probe.cpp)
  target_compile_features(asm_${mode} PRIVATE cxx_std_20)
  target_include_directories(asm_${mode} PRIVATE ../eigen-root-cause/qr/include ../eigen-gap/include)
  target_compile_options(asm_${mode} PRIVATE /O2 /fp:precise /FAs "/Fa${CMAKE_CURRENT_BINARY_DIR}/column_${mode}.asm")
endforeach()
target_compile_definitions(asm_cpp PRIVATE KIBO_QR_ABLATION_CPP_COLUMN=1)
