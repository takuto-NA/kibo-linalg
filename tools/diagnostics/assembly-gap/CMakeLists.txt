cmake_minimum_required(VERSION 3.30)
project(assembly_gap LANGUAGES CXX)
set(EIGEN_SOURCE_DIR "" CACHE PATH "Pinned Eigen")
add_library(kernels OBJECT kernels.cpp)
target_compile_features(kernels PRIVATE cxx_std_20)
target_include_directories(kernels PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/variants/baseline" "${EIGEN_SOURCE_DIR}")
target_compile_options(kernels PRIVATE /O2 /fp:precise /FAs "/Fa${CMAKE_CURRENT_BINARY_DIR}/kernels.asm")
add_executable(micro micro.cpp $<TARGET_OBJECTS:kernels>)
target_compile_features(micro PRIVATE cxx_std_20)
target_compile_options(micro PRIVATE /O2 /fp:precise)

foreach(variant baseline dot panel32 forward fused32 fused_panel32 no_validation inline_finite fused_panel_inline)
 add_executable(${variant} phases.cpp)
 target_compile_features(${variant} PRIVATE cxx_std_20)
 target_include_directories(${variant} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/variants/${variant}" ../../../tests "${EIGEN_SOURCE_DIR}")
 target_compile_definitions(${variant} PRIVATE EIGEN_DONT_PARALLELIZE=1)
 target_compile_options(${variant} PRIVATE /O2 /fp:precise)
endforeach()

add_executable(public_actual phases.cpp)
target_compile_features(public_actual PRIVATE cxx_std_20)
target_include_directories(public_actual PRIVATE ../../../include ../../../tests "${EIGEN_SOURCE_DIR}")
target_compile_definitions(public_actual PRIVATE EIGEN_DONT_PARALLELIZE=1)
target_compile_options(public_actual PRIVATE /O2 /fp:precise)
add_library(public_assembly OBJECT public_assembly.cpp)
target_compile_features(public_assembly PRIVATE cxx_std_20)
target_include_directories(public_assembly PRIVATE ../../../include)
target_compile_options(public_assembly PRIVATE /O2 /fp:precise /FAs "/Fa${CMAKE_CURRENT_BINARY_DIR}/public_finite.asm")

foreach(name before_actual public_solver)
 add_executable(${name} ../solver-locality/probe.cpp)
 target_compile_features(${name} PRIVATE cxx_std_20)
 if(name STREQUAL "before_actual")
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/variants/baseline")
 else()
  target_include_directories(${name} PRIVATE ../../../include)
 endif()
 target_include_directories(${name} PRIVATE ../../../tests "${EIGEN_SOURCE_DIR}")
 target_compile_definitions(${name} PRIVATE EIGEN_DONT_PARALLELIZE=1)
 target_compile_options(${name} PRIVATE /O2 /fp:precise)
endforeach()
