cmake_minimum_required(VERSION 3.30)
project(eigen_root_cause LANGUAGES CXX)
set(EIGEN_SOURCE_DIR "" CACHE PATH "Verified pinned Eigen archive headers")
function(probe name owner source)
  add_executable(${name} ${source})
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_include_directories(${name} PRIVATE ${owner}/include ../eigen-gap/include ../eigen-gap/tests "${EIGEN_SOURCE_DIR}" ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(${name} PRIVATE EIGEN_DONT_PARALLELIZE=1 ${ARGN})
  if(MSVC)
    target_compile_options(${name} PRIVATE /O2 /fp:precise)
  endif()
endfunction()
