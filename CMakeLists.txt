cmake_minimum_required(VERSION 3.30)
project(kibo_linalg VERSION 0.1.0 LANGUAGES CXX)

add_library(kibo_linalg INTERFACE)
add_library(kibo::linalg ALIAS kibo_linalg)
target_compile_features(kibo_linalg INTERFACE cxx_std_20)
target_include_directories(kibo_linalg INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)

option(KIBO_BUILD_EXAMPLES "Build public API examples" ON)
if(KIBO_BUILD_EXAMPLES)
    add_executable(kibo_basic examples/basic.cpp)
    target_link_libraries(kibo_basic PRIVATE kibo::linalg)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(kibo_core_tests tests/core_tests.cpp)
    target_link_libraries(kibo_core_tests PRIVATE kibo::linalg)
    if(MSVC)
        target_compile_options(kibo_core_tests PRIVATE /W4 /permissive-)
    else()
        target_compile_options(kibo_core_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME core COMMAND kibo_core_tests)
    add_executable(kibo_no_runtime_tests tests/no_runtime_tests.cpp)
    target_link_libraries(kibo_no_runtime_tests PRIVATE kibo::linalg)
    if(MSVC)
        target_compile_options(kibo_no_runtime_tests PRIVATE /W4 /permissive- /EHa- /EHs- /EHc- /GR-)
        target_compile_definitions(kibo_no_runtime_tests PRIVATE _HAS_EXCEPTIONS=0)
    else()
        target_compile_options(kibo_no_runtime_tests PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti)
    endif()
    add_test(NAME no_runtime COMMAND kibo_no_runtime_tests)
    add_executable(kibo_allocation_tests tests/allocation_tests.cpp)
    target_link_libraries(kibo_allocation_tests PRIVATE kibo::linalg)
    add_test(NAME allocation COMMAND kibo_allocation_tests)
    set_tests_properties(allocation PROPERTIES SKIP_RETURN_CODE 77)
endif()
