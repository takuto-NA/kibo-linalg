cmake_minimum_required(VERSION 3.30)
project(kibo_linalg VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)

add_library(kibo_linalg INTERFACE)
add_library(kibo::linalg ALIAS kibo_linalg)
set_target_properties(kibo_linalg PROPERTIES EXPORT_NAME linalg)
target_compile_features(kibo_linalg INTERFACE cxx_std_20)
target_include_directories(kibo_linalg INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/kibo_linalgConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/kibo_linalgConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/kibo_linalg)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/kibo_linalgConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(TARGETS kibo_linalg EXPORT kibo_linalgTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT kibo_linalgTargets NAMESPACE kibo:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/kibo_linalg)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/kibo_linalgConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/kibo_linalgConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/kibo_linalg)

option(KIBO_BUILD_EXAMPLES "Build public API examples" ON)
if(KIBO_BUILD_EXAMPLES)
    add_executable(kibo_basic examples/basic.cpp)
    target_link_libraries(kibo_basic PRIVATE kibo::linalg)
    add_executable(kibo_common_2x2 examples/common_2x2_host.cpp)
    target_link_libraries(kibo_common_2x2 PRIVATE kibo::linalg)
    add_executable(kibo_migration examples/migration.cpp)
    target_link_libraries(kibo_migration PRIVATE kibo::linalg)
endif()

include(CTest)
option(KIBO_BUILD_BENCHMARKS "Build reference workloads and benchmarks" ON)
if(KIBO_BUILD_BENCHMARKS)
    add_executable(kibo_lm_reference benchmarks/lm_reference.cpp)
    target_link_libraries(kibo_lm_reference PRIVATE kibo::linalg)
    if(BUILD_TESTING)
        add_test(NAME lm_reference COMMAND kibo_lm_reference)
    endif()
endif()
set(KIBO_EIGEN_INCLUDE_DIR "" CACHE PATH "Fixed Eigen 5.0.1 headers for test-only numerical references")
if(BUILD_TESTING AND KIBO_EIGEN_INCLUDE_DIR)
    add_executable(kibo_numerical_tests tests/numerical_tests.cpp)
    target_link_libraries(kibo_numerical_tests PRIVATE kibo::linalg)
    target_include_directories(kibo_numerical_tests SYSTEM PRIVATE ${KIBO_EIGEN_INCLUDE_DIR})
    add_test(NAME numerical COMMAND kibo_numerical_tests $<$<CONFIG:Debug>:--small>)
    add_test(NAME numerical_column COMMAND kibo_numerical_tests --column-factor $<$<CONFIG:Debug>:--small>)
endif()
if(KIBO_BUILD_BENCHMARKS AND KIBO_EIGEN_INCLUDE_DIR)
    add_executable(kibo_dense_benchmark benchmarks/dense.cpp)
    target_link_libraries(kibo_dense_benchmark PRIVATE kibo::linalg)
    target_include_directories(kibo_dense_benchmark SYSTEM PRIVATE ${KIBO_EIGEN_INCLUDE_DIR})
    target_compile_definitions(kibo_dense_benchmark PRIVATE EIGEN_DONT_PARALLELIZE=1)
    option(KIBO_SCALAR_EIGEN "Disable Eigen SIMD for a separate scalar comparison" OFF)
    if(KIBO_SCALAR_EIGEN)
        target_compile_definitions(kibo_dense_benchmark PRIVATE EIGEN_DONT_VECTORIZE=1 KIBO_DISABLE_SIMD=1)
        if(MSVC)
            message(FATAL_ERROR "The fixed MSVC ignores /Qvec-. Use GCC or Clang for the scalar benchmark with supported vectorization flags.")
        elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            target_compile_options(kibo_dense_benchmark PRIVATE -fno-vectorize -fno-slp-vectorize -ffp-contract=off)
        else()
            target_compile_options(kibo_dense_benchmark PRIVATE -fno-tree-vectorize -fno-tree-slp-vectorize -ffp-contract=off)
        endif()
    endif()
endif()
if(BUILD_TESTING)
    if(KIBO_BUILD_EXAMPLES)
        add_test(NAME common_2x2 COMMAND kibo_common_2x2)
        add_test(NAME migration COMMAND kibo_migration)
    endif()
    add_executable(kibo_core_tests tests/core_tests.cpp)
    target_link_libraries(kibo_core_tests PRIVATE kibo::linalg)
    if(MSVC)
        target_compile_options(kibo_core_tests PRIVATE /W4 /permissive-)
    else()
        target_compile_options(kibo_core_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME core COMMAND kibo_core_tests)
    add_executable(kibo_dynamic_tests tests/dynamic_tests.cpp)
    target_link_libraries(kibo_dynamic_tests PRIVATE kibo::linalg)
    add_test(NAME dynamic COMMAND kibo_dynamic_tests)
    add_executable(kibo_llt_tests tests/llt_tests.cpp)
    target_link_libraries(kibo_llt_tests PRIVATE kibo::linalg)
    add_test(NAME llt COMMAND kibo_llt_tests)
    add_executable(kibo_qr_tests tests/qr_tests.cpp)
    target_link_libraries(kibo_qr_tests PRIVATE kibo::linalg)
    add_test(NAME qr COMMAND kibo_qr_tests)
    add_executable(kibo_qr_refined_tests tests/qr_refined_tests.cpp)
    target_link_libraries(kibo_qr_refined_tests PRIVATE kibo::linalg)
    add_test(NAME qr_refined COMMAND kibo_qr_refined_tests)
    foreach(solver IN ITEMS llt qr qr_refined)
        add_executable(kibo_${solver}_scalar_tests tests/${solver}_tests.cpp)
        target_link_libraries(kibo_${solver}_scalar_tests PRIVATE kibo::linalg)
        target_compile_definitions(kibo_${solver}_scalar_tests PRIVATE KIBO_DISABLE_SIMD=1)
        add_test(NAME ${solver}_scalar COMMAND kibo_${solver}_scalar_tests)
    endforeach()
    add_executable(kibo_no_runtime_tests tests/no_runtime_tests.cpp)
    target_link_libraries(kibo_no_runtime_tests PRIVATE kibo::linalg)
    if(MSVC)
        target_compile_options(kibo_no_runtime_tests PRIVATE /W4 /permissive- /EHa- /EHs- /EHc- /GR-)
        target_compile_definitions(kibo_no_runtime_tests PRIVATE _HAS_EXCEPTIONS=0)
    else()
        target_compile_options(kibo_no_runtime_tests PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti)
    endif()
    add_test(NAME no_runtime COMMAND kibo_no_runtime_tests)
    add_executable(kibo_dynamic_no_runtime_tests tests/dynamic_tests.cpp)
    target_link_libraries(kibo_dynamic_no_runtime_tests PRIVATE kibo::linalg)
    target_compile_options(kibo_dynamic_no_runtime_tests PRIVATE $<TARGET_PROPERTY:kibo_no_runtime_tests,COMPILE_OPTIONS>)
    target_compile_definitions(kibo_dynamic_no_runtime_tests PRIVATE $<TARGET_PROPERTY:kibo_no_runtime_tests,COMPILE_DEFINITIONS>)
    add_test(NAME dynamic_no_runtime COMMAND kibo_dynamic_no_runtime_tests)
    add_executable(kibo_allocation_tests tests/allocation_tests.cpp)
    target_link_libraries(kibo_allocation_tests PRIVATE kibo::linalg)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_compile_definitions(kibo_allocation_tests PRIVATE KIBO_WRAP_ALLOCATORS=1)
        target_compile_options(kibo_allocation_tests PRIVATE -fno-builtin-malloc -fno-builtin-calloc -fno-builtin-realloc -fno-builtin-aligned_alloc)
        target_link_options(kibo_allocation_tests PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=aligned_alloc)
    endif()
    add_test(NAME allocation COMMAND kibo_allocation_tests)
    set_tests_properties(allocation PROPERTIES SKIP_RETURN_CODE 77)
endif()
