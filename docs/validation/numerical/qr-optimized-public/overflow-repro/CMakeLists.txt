cmake_minimum_required(VERSION 3.30)
project(qr_overflow_repro LANGUAGES CXX)
foreach(name IN ITEMS current original unfixed)
  add_executable(${name} main.cpp)
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_compile_options(${name} PRIVATE /O2 /fp:precise)
endforeach()
target_include_directories(current PRIVATE ../../include)
target_include_directories(original PRIVATE ../qr-next/variants/both_finite)
target_include_directories(unfixed PRIVATE ../qr-next/variants/column_work_final)
