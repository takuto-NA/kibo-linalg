cmake_minimum_required(VERSION 3.30)
project(qr_accuracy LANGUAGES CXX)
add_executable(audit audit.cpp)
target_compile_features(audit PRIVATE cxx_std_20)
target_include_directories(audit PRIVATE ../../include ../../tests ../../.cache/eigen)
target_compile_definitions(audit PRIVATE EIGEN_DONT_PARALLELIZE=1)
if(MSVC)
  target_compile_options(audit PRIVATE /O2 /fp:precise)
else()
  target_compile_options(audit PRIVATE -O2 -ffp-contract=off)
endif()
add_executable(cost cost.cpp)
target_compile_features(cost PRIVATE cxx_std_20)
target_include_directories(cost PRIVATE ../../include ../../tests ../../.cache/eigen)
target_compile_definitions(cost PRIVATE EIGEN_DONT_PARALLELIZE=1)
target_compile_options(cost PRIVATE /O2 /fp:precise)
