cmake_minimum_required(VERSION 3.30)
project(solver_locality LANGUAGES CXX)
set(EIGEN_SOURCE_DIR "" CACHE PATH "Pinned Eigen")
set(BASELINE_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/before-include" CACHE PATH "Uninstrumented headers extracted from the baseline commit")
foreach(version before after)
  foreach(solver llt qr_row qr_column)
    set(target ${version}_${solver})
    add_executable(${target} probe.cpp)
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(version STREQUAL "before")
      target_include_directories(${target} PRIVATE "${BASELINE_INCLUDE_DIR}")
    else()
      target_include_directories(${target} PRIVATE ../../../include)
    endif()
    target_include_directories(${target} PRIVATE ../../../tests "${EIGEN_SOURCE_DIR}")
    target_compile_definitions(${target} PRIVATE EIGEN_DONT_PARALLELIZE=1)
    if(NOT solver STREQUAL "llt")
      target_compile_definitions(${target} PRIVATE CAUSE_QR=1)
    endif()
    if(solver STREQUAL "qr_column")
      target_compile_definitions(${target} PRIVATE CAUSE_COLUMN=1)
    endif()
    target_compile_options(${target} PRIVATE /O2 /fp:precise)
  endforeach()
endforeach()
add_executable(before_numerical ../../../tests/numerical_tests.cpp)
target_compile_features(before_numerical PRIVATE cxx_std_20)
target_include_directories(before_numerical PRIVATE "${BASELINE_INCLUDE_DIR}" "${EIGEN_SOURCE_DIR}")
target_compile_options(before_numerical PRIVATE /O2 /fp:precise)
