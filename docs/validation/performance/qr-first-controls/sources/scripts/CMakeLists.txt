cmake_minimum_required(VERSION 3.30)
project(qr_instruction_comparison LANGUAGES CXX)
foreach(name IN ITEMS public_qr_column public_qr_row_same public_qr_row_best)
    add_executable(${name} full-adaptive.cpp)
    target_compile_features(${name} PRIVATE cxx_std_20)
    target_include_directories(${name} PRIVATE variants/baseline ../../tests ../../.cache/eigen)
    target_compile_definitions(${name} PRIVATE EIGEN_DONT_PARALLELIZE=1)
    target_compile_options(${name} PRIVATE /O2 /fp:precise /FAs "/Fa${CMAKE_CURRENT_BINARY_DIR}/${name}.asm")
endforeach()
target_compile_definitions(public_qr_column PRIVATE QR_COLUMN=1)
